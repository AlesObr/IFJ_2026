

<program> ::= import ifj26* EOL <prog_body> EOF
<prog_body> ::= <extension> <prog_body> | <func_def>  <prog_body> | EOL <prog_body> | ε

<extension> ::= extension ( <ext_param> ) EOL [indent] <ext_body> [outdent]
<ext_param> ::= [id] : [type]
<ext_body> ::= <func_def> <ext_tail>
<ext_tail> ::= <func_def> <ext_tail> | EOL <ext_tail> | ε

<func_def> ::= <main_sign> def [id] <param_section>  <data_type> = <func_body>
<func_body> ::= <statement> | EOL <code_block>
<param_section> ::= <param_list> <param_section> | ε
<param_list> ::= ( <param_list_in>
<param_list_in> ::= <param_mem> ) | )
<param_mem> ::= [id] : [type] <param_tail>
<param_tail> ::= , <param_next> | ε
<param_next> ::= [id] : [type] <param_tail> | EOL <param_next> | ε
<main_sign> ::= @main <opt_EOL> | ε
<opt_EOL> ::= EOL | ε

<data_type> ::=  : [type] | ε 

<if_statement> ::= if <expr> then <if_body> <if_tail>
<if_body> ::= EOL <code_block> | <expr>
<if_tail> ::= else <else_if_statement> | ε✓
<else_if_statement> ::= if <expr> then <if_body> <if_tail> | EOL <code_block> <if_tail> | <expr> <if_tail>

<while_statement> ::= while <expr> do EOL <code_block>✓

<code_block> ::= [indent] <code_block_in> [outdent]✓
<code_block_in> ::= <statement> <separator> <statement_tail>✓
<statement_tail> ::= <statement> <separator> <statement_tail> | ε
<separator> ::= EOL | ;
<statement> ::= <if_statement> | <while_statement> | <for_statement> | <return> | <decl> | <expr>
<decl> ::= <decl_type> [id] <data_type> = <expr>
<decl_type> ::= val | var
<return> ::= return <return_val>
<return_val> ::= <expr> | ε

<for_statement> ::= for <for_tail>
<for_tail> ::= <header> do <func_body> | EOL [indent] <header> [outdent] do <func_body>✓
<header> ::= [id] <- <range> <guard_list>✓
<range> ::= <expr> <range_type>✓
<range_type> ::= to <expr> <step> | until <expr> <step>
<step> ::= by <expr> | ε✓
<guard_list> ::= EOL <guard> | ε 
<guard> ::= if <expr> <guard_list> | ε
